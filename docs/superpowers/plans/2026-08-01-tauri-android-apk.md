# Tauri Android APK Support Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build and publish a signed universal Android APK for this Tauri fork, without adding AAB or Google Play distribution.

**Architecture:** `tauri android init` creates the Android Gradle project under `apps/tauri/src-tauri/gen/android`, which becomes fork-owned source. Local scripts build the Android shim and call the Tauri Android CLI through `apps/tauri/run-tauri.mjs`. Release CI stops using `tauri-action`: build jobs upload workflow artifacts, and one final publish job creates the draft GitHub Release and uploads every desktop and Android artifact.

**Tech Stack:** Tauri 2.11, Rust Android targets (`aarch64-linux-android`, `armv7-linux-androideabi`, `i686-linux-android`, `x86_64-linux-android`), Android Gradle project generated by Tauri, GitHub Actions, `actions/upload-artifact@v7`, `actions/download-artifact@v8`, `gh release`.

**Spec:** `docs/superpowers/specs/2026-08-01-tauri-android-design.md`

## Global Constraints

- `packages/ui` stays upstream-owned and is not edited for Android behavior.
- The root `Keystore.jks` is local-only and must not be moved into a tracked path or committed.
- `apps/tauri/src-tauri/gen/android/keystore.properties` is private and gitignored.
- Android releases are APK-only; do not build or upload `.aab` files.
- Release artifacts are created by dedicated build jobs and uploaded by one publish job; do not use `tauri-action`.
- Desktop-only Rust plugins remain gated behind `cfg(not(any(target_os = "android", target_os = "ios")))`.
- Do not hand-edit generated output: `.nuxt/`, `.nitro/`, `.output/`, `packages/ui/.output/`, `apps/tauri/src-tauri/shim.js`, `apps/tauri/src-tauri/target/`, or `apps/tauri/src-tauri/gen/schemas/`.
- Do not hand-edit `pnpm-lock.yaml`; regenerate it with pnpm only if a package manifest dependency change requires it. This plan adds no dependencies.

- [ ] **Step 3: Add keystore loading helpers**

After the `plugins { ... }` block in `apps/tauri/src-tauri/gen/android/app/build.gradle.kts`, add:

```kotlin
val keystorePropertiesFile = rootProject.file("keystore.properties")
val keystoreProperties = Properties()
val hasReleaseSigning = keystorePropertiesFile.exists()

if (hasReleaseSigning) {
    keystoreProperties.load(FileInputStream(keystorePropertiesFile))
}

fun requireKeystoreProperty(name: String): String =
    keystoreProperties.getProperty(name)
        ?: error("Missing Android signing property '$name' in ${keystorePropertiesFile.path}")

fun releaseKeyPassword(): String =
    keystoreProperties.getProperty("keyPassword") ?: requireKeystoreProperty("password")

gradle.taskGraph.whenReady {
    val needsReleaseSigning = allTasks.any { task ->
        task.name.contains("Release", ignoreCase = true)
    }
    if (needsReleaseSigning && !hasReleaseSigning) {
        error("Missing ${keystorePropertiesFile.path}; create it before building a release APK")
    }
}
```

- [ ] **Step 3: Add the `build-desktop` matrix job**

Add this job after `gate`:

```yaml
build-desktop:
  needs: gate

  strategy:
    fail-fast: false
    matrix:
      include:
        - name: linux-x64
          platform: ubuntu-22.04
          args: ''
          rustTargets: ''
        - name: windows-x64
          platform: windows-latest
          args: ''
          rustTargets: ''
        - name: darwin-universal
          platform: macos-latest
          args: '--target universal-apple-darwin'
          rustTargets: 'aarch64-apple-darwin,x86_64-apple-darwin'

  runs-on: ${{ matrix.platform }}

  steps:
    - uses: actions/checkout@v7

    - uses: pnpm/action-setup@v6

    - uses: actions/setup-node@v7
      with:
        cache: pnpm
        node-version: lts/*

    - uses: dtolnay/rust-toolchain@stable
      with:
        targets: ${{ matrix.rustTargets }}

    - uses: Swatinem/rust-cache@v2
      with:
        workspaces: apps/tauri/src-tauri
        key: ${{ matrix.platform }}
        cache-on-failure: true

    - name: Install Linux build dependencies
      if: matrix.platform == 'ubuntu-22.04'
      run: |
        sudo apt-get update
        sudo apt-get install -y \
          libwebkit2gtk-4.1-dev \
          librsvg2-dev \
          patchelf \
          build-essential \
          libdbus-1-dev

    - name: Install dependencies
      run: pnpm install

    - name: Set the bundle version from the tag
      env:
        VERSION: ${{ needs.gate.outputs.version }}
      shell: bash
      run: |
        node -e '
          const fs = require("node:fs")
          const path = "apps/tauri/src-tauri/tauri.conf.json"
          const config = JSON.parse(fs.readFileSync(path, "utf8"))
          const version = process.env.VERSION
          config.version = version
          config.bundle ??= {}
          config.bundle.windows ??= {}
          config.bundle.windows.wix ??= {}
          config.bundle.windows.wix.version = version.split("-")[0]
          fs.writeFileSync(path, JSON.stringify(config, null, 2) + "\n")
        '

    - name: Generate the transport shim
      run: pnpm --filter @metacubexd/tauri build:shim

    - name: Build desktop bundle
      working-directory: apps/tauri
      shell: bash
      run: node run-tauri.mjs build ${{ matrix.args }}

    - name: Collect desktop release artifacts
      env:
        PLATFORM_NAME: ${{ matrix.name }}
        VERSION: ${{ needs.gate.outputs.version }}
      shell: bash
      run: |
        set -euo pipefail
        mkdir -p release-artifacts

        case "$PLATFORM_NAME" in
          linux-x64)
            cp apps/tauri/src-tauri/target/release/bundle/deb/*.deb release-artifacts/
            cp apps/tauri/src-tauri/target/release/bundle/rpm/*.rpm release-artifacts/
            cp apps/tauri/src-tauri/target/release/app release-artifacts/app_linux_x64
            ;;
          windows-x64)
            cp apps/tauri/src-tauri/target/release/bundle/msi/*.msi release-artifacts/
            cp apps/tauri/src-tauri/target/release/bundle/nsis/*.exe release-artifacts/
            cp apps/tauri/src-tauri/target/release/app.exe release-artifacts/app_windows_x64.exe
            ;;
          darwin-universal)
            base="apps/tauri/src-tauri/target/universal-apple-darwin/release"
            cp "$base"/bundle/dmg/*.dmg release-artifacts/
            app_dir="$(find "$base/bundle/macos" -maxdepth 1 -type d -name '*.app' -print -quit)"
            test -n "$app_dir"
            tar -czf "release-artifacts/MetaCubeXD_${VERSION}_universal.app.tar.gz" -C "$(dirname "$app_dir")" "$(basename "$app_dir")"
            cp "$base/app" release-artifacts/app_darwin_universal
            ;;
          *)
            echo "unknown platform $PLATFORM_NAME" >&2
            exit 1
            ;;
        esac

        find release-artifacts -maxdepth 1 -type f -print -quit | grep -q .
        ls -lh release-artifacts

    - uses: actions/upload-artifact@v7
      with:
        name: release-${{ matrix.name }}
        path: release-artifacts/*
        if-no-files-found: error
        retention-days: 5
```

- [ ] **Step 5: Add the single `publish` job**

Add this job after `build-android`:

```yaml
publish:
  needs:
    - gate
    - build-desktop
    - build-android
  runs-on: ubuntu-22.04
  permissions:
    contents: write

  steps:
    - uses: actions/download-artifact@v8
      with:
        pattern: release-*
        path: release-artifacts
        merge-multiple: true

    - name: Publish draft release
      env:
        GITHUB_TOKEN: ${{ secrets.GITHUB_TOKEN }}
        VERSION: ${{ needs.gate.outputs.version }}
        TAG_NAME: ${{ github.ref_name }}
      shell: bash
      run: |
        set -euo pipefail
        find release-artifacts -maxdepth 1 -type f -print -quit | grep -q .
        ls -lh release-artifacts

        prerelease_args=()
        if [[ "$TAG_NAME" == *-rc* ]]; then
          prerelease_args=(--prerelease)
        fi

        if gh release view "$TAG_NAME" >/dev/null 2>&1; then
          gh release upload "$TAG_NAME" release-artifacts/* --clobber
        else
          gh release create "$TAG_NAME" release-artifacts/* \
            --draft \
            "${prerelease_args[@]}" \
            --title "MetaCubeXD $VERSION" \
            --notes "Draft release generated from $TAG_NAME."
        fi
```
